find_package(nlohmann_json REQUIRED CONFIG)
add_library(psim_simulator_model STATIC model.cpp)
target_link_libraries(psim_simulator_model PUBLIC psim::platform_connector)
psim_target_defaults(psim_simulator_model)
add_executable(psim-simulator main.cpp)
target_link_libraries(psim-simulator PRIVATE psim_simulator_model nlohmann_json::nlohmann_json)
target_compile_definitions(psim-simulator PRIVATE PSIM_CONNECTOR_VERSION="${PROJECT_VERSION}")
psim_target_defaults(psim-simulator)
psim_add_test(psim_simulator_test SOURCES tests/model_test.cpp LIBS psim_simulator_model)
install(TARGETS psim-simulator RUNTIME DESTINATION bin COMPONENT simulator)
psim_package(simulator SUMMARY "PSIM reference simulator and event load generator")
