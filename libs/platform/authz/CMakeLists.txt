set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/contracts/authz/actions.json")
find_package(nlohmann_json REQUIRED CONFIG)
file(READ "${PROJECT_SOURCE_DIR}/contracts/authz/actions.json" PSIM_AUTHZ_POLICY)
configure_file(src/policy.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/policy.hpp" @ONLY)
add_library(psim_platform_authz STATIC src/policy.cpp)
add_library(psim::platform_authz ALIAS psim_platform_authz)
target_include_directories(psim_platform_authz PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(psim_platform_authz PUBLIC psim::platform_security PRIVATE nlohmann_json::nlohmann_json)
