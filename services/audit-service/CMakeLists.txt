find_package(OpenSSL REQUIRED CONFIG)
add_library(psim_audit_repository STATIC adapters/chain.cpp adapters/repository.cpp adapters/http.cpp adapters/identity.cpp)
target_include_directories(psim_audit_repository PUBLIC adapters)
target_link_libraries(psim_audit_repository PUBLIC psim::platform_stream psim::platform_authz PRIVATE OpenSSL::Crypto)
psim_target_defaults(psim_audit_repository)
psim_add_service(audit-service SOURCES src/main.cpp LIBS psim_audit_repository SUMMARY "PSIM Audit Service")
add_executable(psim-audit-tool src/tool.cpp)
target_link_libraries(psim-audit-tool PRIVATE psim_audit_repository)
psim_target_defaults(psim-audit-tool)
install(TARGETS psim-audit-tool RUNTIME DESTINATION bin COMPONENT audit-tool)
psim_package(audit-tool SUMMARY "PSIM Audit Verification and Migration Tool")
add_test(NAME psim-audit-tool.version COMMAND psim-audit-tool --version)
add_test(NAME psim-audit-tool.usage COMMAND psim-audit-tool)
set_tests_properties(psim-audit-tool.version psim-audit-tool.usage PROPERTIES LABELS "smoke;psim-audit-tool")
set_tests_properties(psim-audit-tool.usage PROPERTIES WILL_FAIL TRUE)
psim_add_test(psim_audit_test SOURCES tests/chain_test.cpp LIBS psim_audit_repository)
add_executable(psim-audit-probe tests/probe.cpp)
target_link_libraries(psim-audit-probe PRIVATE psim_audit_repository)
psim_target_defaults(psim-audit-probe)
